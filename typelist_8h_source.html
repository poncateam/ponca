80 // General case, head conditionnaly adds current type to the list;

82 struct filter<Pred, T, Ts...>

84 static constexpr bool keepT = Pred::template value<T>;
85 using head = std::conditional_t<keepT, std::tuple<T>, std::tuple<>>;
86 using tail = typename filter<Pred, Ts...>::type;

88 };
