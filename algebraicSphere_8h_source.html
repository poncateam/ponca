100 PONCA_MULTIARCH_STD_MATH(pow);
101 const Scalar squaredEpsilon = epsilon * epsilon;

145 PONCA_MULTIARCH inline void changeBasis(const VectorType& newbasis)
146 {
147 auto& frame = Base::getNeighborFrame();
148
149 VectorType diff = frame.center() - newbasis;
150 frame.changeNeighborhoodFrame(newbasis);
151 Base::init();
152 m_uc = m_uc - m_ul.dot(diff) + m_uq * diff.dot(diff);
153 m_ul = m_ul - Scalar(2.) * m_uq * diff;
154 // m_uq is not changed
155 m_isNormalized = false;
156 applyPrattNorm();
157 }

185 PONCA_MULTIARCH inline bool applyPrattNorm()
186 {
187 if (!m_isNormalized)
188 {
189 Scalar pn = prattNorm();
190 m_uc /= pn;
191 m_ul *= Scalar(1.) / pn;
192 m_uq /= pn;
193
194 m_isNormalized = true;
195 }
196 return true;
197 }

203 PONCA_MULTIARCH [[nodiscard]] inline Scalar radius() const
204 {
205 PONCA_MULTIARCH_CU_STD_FUNC(numeric_limits);
206 if (isPlane())
207 return numeric_limits<Scalar>::infinity(); // non-sense value
208
209 PONCA_MULTIARCH_STD_MATH(sqrt);
210 Scalar b = Scalar(1.) / m_uq;
211 return Scalar(sqrt(((Scalar(-0.5) * b) * m_ul).squaredNorm() - m_uc * b));
212 }

219 }
