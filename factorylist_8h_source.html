18#define DECLARE_FACTORY_CONCEPT_FULL(_conceptname, _newname, _notnewname) \
19 template <typename T> \
20 struct _newname : std::bool_constant<_conceptname<typename T::type>> \
21 { \
22 }; \
23 template <typename T> \
24 struct _notnewname : std::bool_constant<!_conceptname<typename T::type>> \
25 { \
26 };
