100 template <typename _Scalar, unsigned int m, unsigned int n>
101 class PolynomialSmoothWeightKernel
102 {
103 public:
105 using Scalar = _Scalar;
106
108 static constexpr bool isCompact = true;
109
110 // Functor
112 PONCA_MULTIARCH [[nodiscard]] inline Scalar f(const Scalar& _x) const
113 {
114 PONCA_MULTIARCH_STD_MATH(pow);
115 return pow(pow(_x, Scalar(n)) - Scalar(1.), Scalar(m));
116 }
117
119 PONCA_MULTIARCH [[nodiscard]] inline Scalar df(const Scalar& _x) const
120 {
121 PONCA_MULTIARCH_STD_MATH(pow);
122
123 // Check special values that could cause problems at x=0 or x=1
124 if constexpr (m == 0 || n == 0)
125 return Scalar(0);
126
127 return Scalar(m * n) * pow(_x, Scalar(n - 1)) * pow(pow(_x, Scalar(n)) - Scalar(1), Scalar(m - 1));
128 }
129
131 PONCA_MULTIARCH [[nodiscard]] inline Scalar ddf(const Scalar& _x) const
132 {
133 PONCA_MULTIARCH_STD_MATH(pow);
134
135 // Check special values that could cause problems at x=0 or x=1
136 if constexpr (m == 0 || n == 0)
137 return Scalar(0);
138
139 if constexpr (m == 1)
140 {
141 if constexpr (n == 1)
142 return Scalar(0);
143 else
144 return n * (n - 1) * pow(_x, Scalar(n - 2));
145 }
146 else if constexpr (n == 1)
147 return m * (m - 1) * pow(_x - Scalar(1), Scalar(m - 2));
148
149 return Scalar((m - 1) * m * n * n) * pow(_x, Scalar(2 * n - 2)) *
150 pow(pow(_x, Scalar(n)) - Scalar(1), Scalar(m - 2)) +
151 Scalar(m * (n - 1) * n) * pow(_x, Scalar(n - 2)) *
152 pow(pow(_x, Scalar(n)) - Scalar(1), Scalar(m - 1));
153 }
154
155 static constexpr bool isDValid = true;
157 static constexpr bool isDDValid = true;
158 }; // class PolynomialSmoothWeightKernel

244 template <typename _Scalar>
245 class CompactExpWeightKernel
246 {
247 public:
249 using Scalar = _Scalar;
250
252 static constexpr bool isCompact = true;
253
254 // Functor
259 PONCA_MULTIARCH [[nodiscard]] inline Scalar f(const Scalar& _x) const
260 {
261 PONCA_MULTIARCH_STD_MATH(exp);
262 // Avoid divide by 0
263 if (_x >= Scalar(1))
264 return Scalar(0);
265
266 Scalar v = _x * _x;
267 return exp(-v / (Scalar(1) - v));
268 }
269
274 PONCA_MULTIARCH [[nodiscard]] inline Scalar df(const Scalar& _x) const
275 {
276 PONCA_MULTIARCH_STD_MATH(exp);
277 // Avoid divide by 0
278 if (_x >= Scalar(1))
279 return Scalar(0);
280
281 Scalar v = _x * _x;
282 Scalar mv = v - Scalar(1);
283 return -(Scalar(2) * exp(v / mv) * _x) / (mv * mv);
284 }
285
288 PONCA_MULTIARCH [[nodiscard]] inline Scalar ddf(const Scalar& _x) const
289 {
290 PONCA_MULTIARCH_STD_MATH(exp);
291 // Avoid divide by 0
292 if (_x >= Scalar(1))
293 return Scalar(0);
294
295 Scalar v = _x * _x;
296 Scalar mv = v - Scalar(1);
297 return Scalar(2) * exp(v / mv) * (Scalar(3) * v * v - Scalar(1)) / (mv * mv * mv * mv);
298 }
299
301 static constexpr bool isDValid = true;
303 static constexpr bool isDDValid = true;
304 }; // class CompactExpWeightKernel
