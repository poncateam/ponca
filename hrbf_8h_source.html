81 template <typename IteratorBegin, typename IteratorEnd>
82 inline FIT_RESULT compute(const IteratorBegin& begin, const IteratorEnd& end)
83 {
84 constexpr int DIM = DataPoint::Dim;
85
86 const int nb_points = std::distance(begin, end);
87 const int nb_hrbf_constraints = (DIM + 1) * nb_points;
88 const int nb_constraints = nb_hrbf_constraints;
89 const int nb_coeffs = (DIM + 1) * nb_points;
90 const WeightKernel wk;
91
92 m_node_centers.resize(DIM, nb_points);
93 m_betas.resize(DIM, nb_points);
94 m_alphas.resize(nb_points);
95
96 // Assemble the "design" and "value" matrix and vector
97 DenseMatrix D(nb_constraints, nb_coeffs);
98 DenseVector f(nb_constraints);
99
100 // copy the node centers
101 auto it = begin;
102 for (int i = 0; i < nb_points; ++i, ++it)
103 m_node_centers.col(i) = (*it).pos();
104
105 it = begin;
106 for (int i = 0; i < nb_points; ++i, ++it)
107 {
108 const auto& p = (*it).pos();
109 const auto& n = (*it).normal();
110
111 const int io = (DIM + 1) * i;
112 f(io) = 0;
113 f.template segment<DIM>(io + 1) = n;
114
115 for (int j = 0; j < nb_points; ++j)
116 {
117 const int jo = (DIM + 1) * j;
118 const VectorType diff = p - m_node_centers.col(j);
119 const Scalar l = diff.norm();
120
121 if (l < eps)
122 {
123 D.template block<DIM + 1, DIM + 1>(io, jo).setZero();
124 }
125 else
126 {
127 const Scalar w = wk.f(l);
128 const Scalar dw_l = wk.df(l) / l;
129 const Scalar ddw = wk.ddf(l);
130 const VectorType g = diff * dw_l;
131
132 D(io, jo) = w;
133 D.row(io).template segment<DIM>(jo + 1) = g.transpose();
134 D.col(jo).template segment<DIM>(io + 1) = g;
135 D.template block<DIM, DIM>(io + 1, jo + 1) = (ddw - dw_l) / (l * l) * (diff * diff.transpose());
136 D.template block<DIM, DIM>(io + 1, jo + 1).diagonal().array() += dw_l;
137 }
138 }
139 }
140
141 const DenseVector x = D.lu().solve(f);
142 Eigen::Map<const Eigen::Matrix<Scalar, DIM + 1, Eigen::Dynamic>> mx(x.data(), DIM + 1, nb_points);
143
144 m_alphas = mx.row(0);
145 m_betas = mx.template bottomRows<DIM>();
146
147 // Catches cases where the system is ill-conditionned (duplicate / 'colinear' points)
148 if (!x.allFinite() || (D * x - f).norm() > std::sqrt(eps) * f.norm())
149 return m_eCurrentState = UNSTABLE;
150
151 return m_eCurrentState = STABLE;
152 }

174 [[nodiscard]] inline Scalar potential(const VectorType& x) const
175 {
176 const int nb_nodes = m_node_centers.cols();
177 const WeightKernel wk;
178
179 Scalar ret = 0;
180 for (int i = 0; i < nb_nodes; ++i)
181 {
182 const VectorType diff = x - m_node_centers.col(i);
183 const Scalar l = diff.norm();
184
185 if (l > eps)
186 {
187 ret += m_alphas(i) * wk.f(l);
188 ret += m_betas.col(i).dot(diff) * wk.df(l) / l;
189 }
190 }
191 return ret;
192 }

194 [[nodiscard]] inline VectorType primitiveGradient(const VectorType& x) const
195 {
196 const int nb_nodes = m_node_centers.cols();
197 const WeightKernel wk;
198
199 VectorType grad = VectorType::Zero();
200 for (int i = 0; i < nb_nodes; i++)
201 {
202 const VectorType node = m_node_centers.col(i);
203 const VectorType beta = m_betas.col(i);
204 const VectorType diff = x - node;
205 const Scalar alpha = m_alphas(i);
206 const Scalar l = diff.norm();
207
208 if (l > eps)
209 {
210 const VectorType diffNormalized = diff / l;
211
212 const Scalar dphi = wk.df(l);
213 const Scalar ddphi = wk.ddf(l);
214
215 const Scalar alpha_dphi = alpha * dphi;
216
217 const Scalar bDotd_l = beta.dot(diff) / l;
218 const Scalar squared_l = diff.squaredNorm();
219
220 grad += alpha_dphi * diffNormalized;
221 grad += bDotd_l * (ddphi * diffNormalized - diff * dphi / squared_l) + beta * dphi / l;
222 }
223 }
224 return grad;
225 }
