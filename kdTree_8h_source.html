470 struct DefaultConverter
471 {
472 template <typename Input>
473 PONCA_MULTIARCH_HOST inline void operator()(Input&& i, PointContainer& o)
474 {
475 using InputContainer = std::remove_reference_t<Input>;
476 if constexpr (std::is_same_v<InputContainer, PointContainer> && std::is_copy_assignable_v<DataPoint>)
477 o = std::forward<Input>(i); // Either move or copy
478 else
479 std::transform(
480 i.cbegin(), i.cend(), std::back_inserter(o),
481 [](const typename InputContainer::value_type& p) -> DataPoint { return DataPoint(p); });
482 }
483 };
484
488 template <typename PointUserContainer>
489 PONCA_MULTIARCH_HOST inline void build(PointUserContainer&& points)
490 {
491 build(std::forward<PointUserContainer>(points), DefaultConverter());
492 }
493
494 // Internal ----------------------------------------------------------------
495 protected:
503 template <typename PointUserContainer, typename IndexUserContainer, typename PointConverter>
504 PONCA_MULTIARCH_HOST inline void buildWithSampling(PointUserContainer&& points, IndexUserContainer&& sampling,
505 PointConverter c);
506
512 template <typename PointUserContainer, typename IndexUserContainer>
513 PONCA_MULTIARCH_HOST inline void buildWithSampling(PointUserContainer&& points, IndexUserContainer&& sampling)
514 {
515 buildWithSampling(std::forward<PointUserContainer>(points), std::forward<IndexUserContainer>(sampling),
516 DefaultConverter());
517 }
518
519 private:
520 PONCA_MULTIARCH_HOST inline void buildRec(NodeIndexType node_id, IndexType start, IndexType end, int level);
521 PONCA_MULTIARCH_HOST [[nodiscard]] inline IndexType partition(IndexType start, IndexType end, int dim,
522 Scalar value);
523 };
